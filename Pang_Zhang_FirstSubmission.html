import humidity

temp <- read_excel(path = "./temperature.xlsx", sheet = 1, col_names = TRUE)

temp_daily <- temp %>%
  mutate(date = as.Date(date)) %>%
  group_by(date) %>%
  summarise(daily_mean_temp = mean(c_across(t_ws1:t_ws28), na.rm = TRUE)) %>%
  ungroup()

load_daily_combined <- load_daily %>%
  left_join(temp_daily, by = "date")

humidity <- read_excel(path = "./relative_humidity.xlsx", sheet = 1, col_names = TRUE)

humidity_daily <- humidity %>%
  mutate(date = as.Date(date)) %>%
  group_by(date) %>%
  summarise(daily_mean_rh = mean(c_across(rh_ws1:rh_ws28), na.rm = TRUE)) %>%
  ungroup()

load_daily_combined2 <- load_daily_combined %>%
  left_join(humidity_daily, by = "date")

# xreg
temp_rh_train <- load_daily_combined2 %>%
  filter(date <= as.Date("2010-06-30")) %>%
  pull(daily_mean_temp) %>%
  cbind(load_daily_combined2 %>%
    filter(date <= as.Date("2010-06-30")) %>%
    pull(daily_mean_rh))

temp_rh_test <- load_daily_combined2 %>%
  filter(date >= as.Date("2010-07-01") & date <= as.Date("2010-07-31")) %>%
  pull(daily_mean_temp) %>%
  cbind(load_daily_combined2 %>%
    filter(date >= as.Date("2010-07-01") & date <= as.Date("2010-07-31")) %>%
    pull(daily_mean_rh))

# ARIMA + Fourier + Temp + Humidity
ARIMA_TRH_fit <- Arima(load_ts_train,
                       order = c(2, 0, 1),
                       xreg = cbind(fourier(load_ts_train, K = c(2, 6)), temp_rh_train))

ARIMA_TRH_for <- forecast(ARIMA_TRH_fit,
                          h = n_for,
                          xreg = cbind(fourier(load_ts_train, K = c(2, 6), h = n_for), temp_rh_test))

ARIMA_TRH_scores <- accuracy(ARIMA_TRH_for$mean, load_ts_test)

  select(date, daily_mean) %>%
  rename(ds = date, y = daily_mean) %>%
  filter(ds <= as.Date("2010-06-30"))

prophet_fit <- prophet(prophet_df,
                       weekly.seasonality = TRUE,
                       yearly.seasonality = TRUE,
                       daily.seasonality = FALSE)

future <- make_future_dataframe(prophet_fit, periods = 31)
prophet_for <- predict(prophet_fit, future)

prophet_pred <- tail(prophet_for$yhat, 31)

prophet_scores <- accuracy(prophet_pred, load_ts_test)
